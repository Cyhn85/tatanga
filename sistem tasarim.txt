def calculate_trend_score(product):
    """
    Her ürün için trend skoru hesapla
    """
    
    score = 0
    
    # Favori sayısı (ağırlık: %40)
    score += product.favorites * 4
    
    # Görüntülenme sayısı (ağırlık: %20)
    score += product.views * 0.2
    
    # Satış hızı (ağırlık: %30)
    # Son 7 günde kaç benzer ürün satılmış
    similar_sold = count_similar_sold(product, days=7)
    score += similar_sold * 3
    
    # Rekabet durumu (ağırlık: %10)
    # Az satıcı = yüksek puan
    competitors = count_competitors(product)
    if competitors < 5:
        score += 50
    elif competitors < 10:
        score += 30
    
    return score

def match_products_across_platforms(search_term):
    """
    Aynı ürünü farklı platformlarda bul
    """
    
    results = {
        "vinted": [],
        "kleinanzeigen": [],
        "ebay": [],
        "willhaben": [],
    }
    
    # Her platformdan ara
    for platform in results.keys():
        products = scan_platform(platform, search_term)
        results[platform] = products
    
    # Eşleştir
    matches = []
    for v_product in results["vinted"]:
        for k_product in results["kleinanzeigen"]:
            similarity = calculate_similarity(v_product, k_product)
            
            if similarity > 0.85:  # %85 benzerlik
                price_diff = k_product.price - v_product.price
                
                if price_diff >= 5:  # Min 5€ fark
                    matches.append({
                        "buy_from": "vinted",
                        "buy_price": v_product.price,
                        "sell_on": "kleinanzeigen",
                        "sell_price": k_product.price,
                        "profit": price_diff,
                        "vinted_url": v_product.url,
                        "klein_url": k_product.url,
                    })
    
    return matches

Kar Hesaplama (Kargo Dahil)
def calculate_real_profit(opportunity):
    """
    Gerçek kar hesapla - kargo dahil
    """
    
    buy_price = opportunity.buy_price
    sell_price = opportunity.sell_price
    
    # Kargo maliyetleri
    if opportunity.buy_platform == "vinted":
        # Vinted anlaşmalı kargo
        buy_shipping = get_vinted_shipping_cost(
            from_location=opportunity.seller_location,
            to_location="Berlin"
        )
    else:
        buy_shipping = estimate_shipping(
            opportunity.buy_platform,
            opportunity.seller_location
        )
    
    # Satış kargo maliyeti
    if opportunity.sell_platform == "kleinanzeigen":
        # Alıcıya direkt gönderim seçeneği
        sell_shipping = 0  # Alıcı öder
    else:
        sell_shipping = estimate_shipping(
            opportunity.sell_platform,
            "Berlin"
        )
    
    # Platform komisyonları
    platform_fees = calculate_platform_fees(
        opportunity.sell_platform,
        sell_price
    )
    
    # Net kar
    net_profit = sell_price - buy_price - buy_shipping - sell_shipping - platform_fees
    
    return {
        "gross_profit": sell_price - buy_price,
        "buy_shipping": buy_shipping,
        "sell_shipping": sell_shipping,
        "platform_fees": platform_fees,
        "net_profit": net_profit,
        "roi": (net_profit / buy_price) * 100,
    }

def analyze_seller(seller_profile):
    """
    Satıcıyı analiz et:
    - Cevap verme hızı
    - Pazarlık yapıyor mu
    - Güvenilir mi
    """
    
    analysis = {
        "response_rate": seller_profile.response_rate,  # Vinted'den
        "response_time": seller_profile.avg_response_time,
        "rating": seller_profile.rating,
        "total_sales": seller_profile.sold_items,
        "active_listings": seller_profile.active_items,
    }
    
    # İndirim potansiyeli
    if seller_profile.active_items > 20:
        analysis["discount_potential"] = "high"  # Çok ürünü var, indirim yapar
    elif seller_profile.last_login > 7:  # 7 gündür girmemiş
        analysis["discount_potential"] = "medium"  # Acele edebilir
    else:
        analysis["discount_potential"] = "low"
    
    return analysis

def direct_arbitrage(opportunity, buyer_order):
    """
    Ürünü kendimize almadan direkt alıcıya gönder
    """
    
    # 1. Alıcıdan sipariş geldi (Kleinanzeigen/eBay)
    buyer_address = buyer_order.shipping_address
    
    # 2. Satıcıya (Vinted) alıcının adresini ver
    seller_message = f"""
    Hallo! Ich möchte das Produkt kaufen.
    
    Bitte senden Sie es direkt an diese Adresse:
    
    {buyer_address.name}
    {buyer_address.street}
    {buyer_address.zip} {buyer_address.city}
    {buyer_address.country}
    
    Ich bezahle sofort!
    """
    
    # 3. Ödemeyi yap
    pay_for_product(opportunity.buy_platform, opportunity.product_id)
    
    # 4. Kargo takip numarasını al ve alıcıya ilet
    tracking = await wait_for_tracking(opportunity.order_id)
    
    send_to_buyer(buyer_order.id, f"""
    Ihre Bestellung wurde versandt!
    
    Sendungsnummer: {tracking}
    Lieferzeit: 3-5 Werktage
    """)

15 Dakikada Bir Kontrol
async def periodic_check():
    """
    Her 15 dakikada bir çalışır
    """
    
    while True:
        # Tüm mağazaları kontrol et
        for store in get_all_stores():
            # Mesajları kontrol et
            new_messages = await check_messages(store)
            for msg in new_messages:
                await process_new_message(msg)
            
            # Satışları kontrol et
            new_sales = await check_sales(store)
            for sale in new_sales:
                await process_new_sale(sale)
            
            # Linkleri kontrol et
            broken_links = await check_listing_links(store)
            for link in broken_links:
                await notify_broken_link(link)
        
        # 15 dakika bekle
        await asyncio.sleep(900)

def handle_price_offer(offer):
    """
    Fiyat teklifini değerlendir ve karşılık ver
    """
    
    product = get_product(offer.product_id)
    
    # Minimum fiyat kontrolü
    min_acceptable = product.price * 0.85  # %15 max indirim
    
    if offer.amount >= product.price:
        # Tam fiyat veya üstü - hemen kabul
        action = "accept"
        
    elif offer.amount >= min_acceptable:
        # Kabul edilebilir - küçük karşı teklif
        counter = product.price * 0.95
        action = f"counter:{counter}"
        
    elif offer.amount >= product.price * 0.75:
        # Düşük ama pazarlık yapılabilir
        counter = product.price * 0.90
        action = f"counter:{counter}"
        
    else:
        # Çok düşük - reddet
        action = "reject"
    
    # Platform'a göre işlem
    if offer.platform == "vinted":
        # Vinted buton ile
        execute_vinted_offer_action(offer.id, action)
    elif offer.platform == "ebay":
        # eBay Preisvorschlag
        execute_ebay_offer_action(offer.id, action)
    elif offer.platform == "kleinanzeigen":
        # Mesaj ile
        send_negotiation_message(offer.conversation_id, action)
    
    # Telegram'a bildir
    send_telegram(
        channel="#pazarlik",
        text=f"💰 Pazarlık İşlendi\n\n"
             f"Ürün: {product.title}\n"
             f"Fiyat: {product.price}€\n"
             f"Teklif: {offer.amount}€\n"
             f"Aksiyon: {action}"
    )

# Kullanıcı #kargo odasına yazar:
# "CSS-001 DHL 123456789"

def handle_shipping_command(message):
    """
    Kargo bilgisi güncelle
    """
    
    parts = message.text.split()
    product_id = parts[0]   # CSS-001
    carrier = parts[1]      # DHL
    tracking = parts[2]     # 123456789
    
    # Veritabanı güncelle
    product = get_product(product_id)
    product.shipping = {
        "carrier": carrier,
        "tracking": tracking,
        "status": "shipped",
        "shipped_at": now(),
    }
    save_product(product)
    
    # Alıcıya bildir (platform'a göre)
    if product.sold_on == "kleinanzeigen":
        # Mesaj ile
        send_kleinanzeigen_message(
            conversation_id=product.sale.conversation_id,
            text=f"Ihr Paket wurde versandt!\n"
                 f"Sendungsnummer: {tracking}\n"
                 f"Verfolgen: https://www.dhl.de/tracking?nummer={tracking}"
        )
    elif product.sold_on == "ebay":
        # API ile
        update_ebay_tracking(product.sale.order_id, carrier, tracking)
    
    # Onay
    reply(f"✅ Kargo bilgisi güncellendi\n"
          f"📦 {carrier}: {tracking}")
